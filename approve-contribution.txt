  log(`목록 ${total}건 / 체크 방법: ${api ? '표 기능(' + apiWhere + ')' : '체크칸 클릭'}`);
  if (!(api && apiInfo.getValues)) { log('표를 오른쪽으로 넘기며 팀 이름을 읽는 중…'); await readAcross(); }
  // 팀 확인: 팀 글자가 하나도 안 보이면 패션2팀을 가려낼 수 없으니 멈춘다
  const texts = [...Array(total).keys()].map(rowText);
  if (!texts.some(hasTeam)) { log('목록에서 팀 이름을 못 찾았어요. 패션2팀 건을 가려낼 수 없어서 승인하지 않고 멈췄어요.'); showDiag('팀 칸 없음', [`첫 줄 값: ${texts[0].slice(0, 600)}`]); finish(); return; }
  const skipN = texts.filter((t) => SKIP_TEAM.test(t) || !hasTeam(t)).length;
  const goN = total - skipN;
  const unknownN = texts.filter((t) => !SKIP_TEAM.test(t) && !hasTeam(t)).length;
  log(`패션2팀 ${skipN - unknownN}건${unknownN ? `, 팀 이름을 못 읽은 ${unknownN}건` : ''}은 빼고 ${goN}건 승인 예정`);
  if (!goN) { log('승인할 건이 없어요.'); finish(); return; }
  if (!confirm(`목록 ${total}건 중 패션2팀 ${skipN - unknownN}건${unknownN ? `, 팀 이름을 못 읽은 ${unknownN}건` : ''}은 빼고, ${goN}건을 한 건씩 차례로 승인할까요?`)) { log('취소했어요.'); finish(); return; }

  const nat = { alert: window.alert, confirm: window.confirm };
  window.alert = (m) => log('  창: ' + m);
  window.confirm = (m) => { log('  창: ' + m + ' → 확인'); return true; };
  const done = new Set();
  let ok = 0;
  try {
    for (let n = 0; n < goN && !stop; n++) {
      // 아직 안 한 첫 줄 (승인된 줄은 목록에서 빠지거나, 남아 있어도 건너뜀)
      const cnt = count();
      let i = -1;
      for (let k = 0; k < cnt; k++) {
        const key = rowKey(k);
        if (done.has(key)) continue;
        const tt = rowText(k);
        if (SKIP_TEAM.test(tt)) { done.add(key); log(`  건너뜀 (패션2팀): ${rowLabel(k)}`); continue; }
        if (!hasTeam(tt)) { done.add(key); log(`  건너뜀 (팀 이름을 못 읽음): ${rowLabel(k)}`); continue; }
        i = k; break;
      }
      if (i < 0) { log('남은 건이 없어요.'); break; }
      const key = rowKey(i);
      log(`${n + 1}/${goN} ${rowLabel(i)}`); setProg(n, goN); setMsg(`승인 중… ${n + 1}/${goN}건째`);
      if (api) safe(() => api.setCurrent && api.setCurrent({ itemIndex: i }));
      const how = await checkOnly(i);
      if (!how) { log('  줄 체크가 확인되지 않아 [승인]을 누르지 않고 멈췄어요.'); showDiag('체크 실패', [`대상 줄 ${i}`]); break; }
      const b = approveBtn();
      if (!b) { log('  [승인] 버튼을 못 찾아서 멈춤'); showDiag('승인 버튼 없음'); break; }
      if (b.disabled || b.getAttribute('aria-disabled') === 'true') { log('  [승인]이 꺼져 있어서 멈춤 (권한 확인 필요)'); break; }
      // 세일즈원 확인창은 미리 숨겨 두었다가 보여 주는 방식이라, '새로 생긴 버튼'이 아니라 '지금 맨 위에 보이는 [확인]'으로 찾는다 (2026-10-16 실제 화면)
      const shownOk = () => okButtons().filter((x) => onTop(x));
      const already = new Set(shownOk()); // 누르기 전부터 떠 있던 창은 건드리지 않는다
      press(b);
      let oks = 0, blocked = false;
      for (let round = 0; round < 3; round++) {
        // 확인창이 뜨면 바로 누른다. 첫 창은 최대 5초, 다음 창은 최대 3초 기다린다
        let okb = null;
        for (let w = 0; w < (round ? 3000 : 5000) && !okb; w += 100) {
          await sleep(100);
          okb = shownOk().find((x) => !already.has(x)) || null;
        }
        if (!okb) break;
        await sleep(200); // 창 글자가 다 그려지도록 잠깐
        let msg = '', dlg = null;
        for (let p = okb.parentElement, k = 0; p && p !== document.body && k < 6 && !msg; p = p.parentElement, k++) {
          const c = p.cloneNode(true); c.querySelectorAll('button, [role=button]').forEach((x) => x.remove());
          const tx = norm(c.textContent);
          if (tx.length > 200) break;
          if (tx.length > 3) { msg = tx; dlg = p; }
        }
        // 승인 사유 같은 입력칸이 있는 창이면 비워 둔 채 누르지 않는다
        const field = dlg && [...dlg.querySelectorAll('textarea, input[type=text], input:not([type])')].find((x) => vis(x) && !x.value);
        if (field) { log('  창: ' + msg + ' → 입력칸이 있어서 멈췄어요. 직접 입력해 주세요.'); blocked = true; break; }
        log('  창: ' + msg + ' → 확인');
        press(okb); oks++;
        // 창이 닫힐 때까지 기다린다 (같은 창이 다음 안내에 다시 쓰일 수 있어서)
        for (let w = 0; w < 2000 && onTop(okb); w += 100) await sleep(100);
      }
      if (blocked) { stop = true; break; }
      done.add(key);
      if (oks) { ok++; setProg(n + 1, goN); log('  승인 눌렀어요'); await sleep(1000); } // 목록이 다시 조회되는 동안 잠깐
      else { log('  [승인]을 눌렀는데 확인창을 못 찾아서 멈췄어요.'); showDiag('확인창 못 찾음', [`보이는 [확인]류 버튼: ${okButtons().map((x) => dsc(x) + (onTop(x) ? '(맨 위)' : '(가려짐)')).join(' | ') || '없음'}`, `떠 있는 창 글자: ${[...document.querySelectorAll('[role=dialog], [class*=modal], [class*=Modal], [class*=dialog], [class*=Dialog], [class*=popup], [class*=Popup]')].filter(vis).map((d) => norm(d.textContent).slice(0, 120)).join(' / ') || '없음'}`]); break; }
    }
  } finally { window.alert = nat.alert; window.confirm = nat.confirm; }
  log(`끝. ${goN}건 중 ${ok}건 승인했어요. (${skipN}건 제외)`); if (ok < goN) setMsg(`멈췄어요. ${goN}건 중 ${ok}건 승인했어요. 자세한 기록을 확인해 주세요.`, 'bad'); finish();
})();
