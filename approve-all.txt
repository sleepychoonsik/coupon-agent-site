      try { localStorage.setItem('cpnApproveAll.watch', String(watchMin)); } catch {}
      log(watchMin ? `자동으로 다시 찾기: ${watchMin}분마다` : '자동으로 다시 찾기 끔'); $('#cpn-next').textContent = watchText();
    };
    if ($('#cpn-stop')) $('#cpn-stop').onclick = () => { stop = true; $('#cpn-stop').textContent = '멈추는 중…'; $('#cpn-stop').disabled = true; log('멈추기 누름'); };
  }
  // ---------- 다시 조회해서 새 쿠폰 승인 (사용자 요청 2026-10-22) ----------
  // 승인한 쿠폰은 목록에서 빠지고, 건너뛴 쿠폰(예: 본인이 등록한 쿠폰)은 남는다. 이미 본 쿠폰(seen)은 다시 하지 않는다.
  let watchMin = 0, nextAt = 0, checked = 0, lastNote = '', watchTimer = 0;
  const hhmm = (t) => new Date(t).toTimeString().slice(0, 5);
  const watchText = () => [checked ? `마지막 확인 ${hhmm(checked)}${lastNote ? ' · ' + lastNote : ''}` : '', watchMin && nextAt ? `다음 확인 ${hhmm(nextAt)}` : ''].filter(Boolean).join(' · ');
  async function again(byHand) {
    if (running) return;
    const btn = searchButton();
    if (!btn) { lastNote = '[조회] 버튼을 못 찾았어요'; log('  ' + lastNote); checked = Date.now(); render(); return; }
    await clearSearch();
    log(byHand ? '다시 조회' : '자동으로 다시 조회');
    press(btn);
    await sleep(1200);
    await waitList(10000);
    checked = Date.now();
    if (watchMin) nextAt = Date.now() + watchMin * 60000;
    // 화면 '총 N건' 중 건너뛴 쿠폰을 뺀 것이 새 쿠폰. 표가 보이는 줄만 그리므로 숫자로 먼저 보고, 보이는 줄에 새 코드가 있는지도 본다
    const skippedLeft = items.filter((x) => x.st === 'skip').length;
    const fresh = Math.max(0, totalOnPage() - skippedLeft);
    if (!fresh && !listCodes().some((c) => !seen.has(c))) { lastNote = '새 쿠폰 없음'; log('  새 쿠폰 없음'); render(); return; }
    lastNote = `새 쿠폰 ${fresh || '?'}건`; log('  ' + lastNote);
    await run(fresh || listCodes().filter((c) => !seen.has(c)).length);
  }
  try { watchMin = Number(localStorage.getItem('cpnApproveAll.watch')) || 0; } catch {}
  watchTimer = setInterval(() => {
    if (!box.isConnected) { clearInterval(watchTimer); return; }
    if (!watchMin || running || !(items.length || checked)) return;
    if (!nextAt) nextAt = Date.now() + watchMin * 60000;
    if (Date.now() >= nextAt) { nextAt = Date.now() + watchMin * 60000; again(false); }
  }, 1000);
  render();

  async function run(more) {
  stop = false;
  // 처음엔 화면의 '총 N건', 다시 조회할 땐 이미 한 쿠폰 + 새 쿠폰 수
  startTotal = more ? seen.size + more : (totalOnPage() || listCodes().length);
  running = true; render();
  // 승인하는 동안 브라우저 기본 확인창은 [확인]으로 처리
  const nat = { alert: window.alert, confirm: window.confirm };
  window.alert = (m) => log('  창: ' + m);
  window.confirm = (m) => { log('  창: ' + m + ' → 확인'); return true; };
  try {
    for (let i = 0; i < 1000; i++) {
      if (stop) break;
      await clearSearch();
      const code = await nextCode();
      if (!code) break;
      seen.add(code);
      const it = { code, desc: descOf(code), st: 'run', why: '' };
      items.push(it);
      const skip = (why) => { it.st = 'skip'; it.why = why; log('  ' + why); render(); };
      render();
      log(`${i + 1}. ${code}`);
      // 1) 목록 줄을 눌러 연다 2) 안 열리면 위쪽 쿠폰코드 칸에 이 코드를 넣고 [조회]해서 연다
      let opened = await openByRow(code);
      if (!opened && !stop) { log('  줄을 눌러도 안 열려서 쿠폰코드로 조회해요'); opened = await openBySearch(code); }
      if (!opened) { skip('쿠폰상세에 안 열렸어요'); continue; }
      const b = approveBtn();
      if (!b) { skip('[승인] 버튼을 못 찾아서 멈췄어요'); break; }
      if (b.disabled || b.getAttribute('aria-disabled') === 'true') { skip('[승인]이 꺼져 있어요 (이미 승인됐거나 권한 없음)'); continue; }
      const before = new Set(okButtons());
      if (openedCode() !== code) { skip('누르기 직전에 다른 쿠폰이 열려서 건너뛰었어요'); continue; }
      press(b);
      let oks = 0, err = '';
      for (let round = 0; round < 3; round++) {
        // 확인창이 뜨면 바로 누른다 (사용자 요청 2026-10-12). 첫 창은 최대 5초, 다음 창은 최대 3초 기다린다
        let ok = null;
        for (let w = 0; w < (round ? 3000 : 5000) && !ok; w += 100) {
          await sleep(100);
          ok = okButtons().find((x) => !before.has(x) && onTop(x)) || null;
        }
        if (!ok) break;
        await sleep(100); // 창 글자가 다 그려지도록 잠깐
        const msg = dialogText(ok);
        log('  창: ' + msg + ' → 확인');
        press(ok); oks++; before.add(ok);
        // 승인이 안 된 창 (2026-10-14 실제: '등록자와 승인자는 동일할 수 없습니다.' 를 승인으로 셌음)
        if (BAD.test(msg) && !/하시겠/.test(msg)) { err = msg; break; }
        // '승인되었습니다' 같은 완료 창이면 다음 창을 기다리지 않는다 (사용자 요청 2026-10-13: 확인 → 다음 승인 사이 줄이기)
        if (/되었|완료/.test(msg) && !/하시겠/.test(msg)) break;
      }
      // 목록이 다시 조회되어 승인한 줄이 빠질 때까지만 기다린다 (최대 1.5초)
      if (err) { skip('승인 안 됨: ' + err); continue; }
      if (oks) for (let w = 0; w < 1500 && codeCell(code); w += 100) await sleep(100);
      if (oks) { it.st = 'ok'; log('  승인 눌렀어요'); render(); } else skip('확인창이 안 떴어요. 화면을 확인해 주세요');
    }
  } catch (e) { log('오류로 멈췄어요: ' + (e && e.message ? e.message : e)); }
  finally { window.alert = nat.alert; window.confirm = nat.confirm; try { await clearSearch(); } catch {} running = false; }
  for (const x of items) if (x.st === 'run') { x.st = 'skip'; x.why = x.why || '멈춰서 안 했어요'; }
  log('끝');
  render();
  }
})();
