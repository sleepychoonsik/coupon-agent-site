  // 좌표를 실은 눌림 (그리드는 마우스 좌표로 칸을 찾는다)
  function pressAt(x, y, dbl) {
    const pe = host.style.pointerEvents; host.style.pointerEvents = 'none'; // 도우미 창은 빼고 그 자리의 요소를 찾는다
    const el = document.elementFromPoint(x, y); host.style.pointerEvents = pe;
    if (!el) return null;
    fireMouse(el, x, y, dbl);
    return el;
  }
  // 요소 자체에 좌표를 실어 누른다 (도우미 창이나 다른 요소가 그 자리를 덮고 있어도 그 요소로 간다)
  // 진짜 마우스처럼 포인터 신호(pointerId·pointerType 포함)와 마우스 신호를 차례로 보낸다 (RealGrid는 포인터 신호의 속성을 본다)
  function fireMouse(el, x, y, dbl) {
    const base = { bubbles: true, cancelable: true, composed: true, view: window, button: 0, clientX: x, clientY: y, screenX: x, screenY: y, detail: 1 };
    const pe = (t, extra) => { try { el.dispatchEvent(new PointerEvent(t, { ...base, pointerId: 1, pointerType: 'mouse', isPrimary: true, width: 1, height: 1, pressure: t === 'pointerdown' ? 0.5 : 0, ...extra })); } catch (e) { el.dispatchEvent(new MouseEvent(t, { ...base, ...extra })); } };
    const me = (t, extra) => el.dispatchEvent(new MouseEvent(t, { ...base, ...extra }));
    pe('pointerover', { buttons: 0 }); pe('pointerenter', { bubbles: false, buttons: 0 }); me('mouseover', { buttons: 0 }); me('mousemove', { buttons: 0 });
    pe('pointerdown', { buttons: 1 }); me('mousedown', { buttons: 1 });
    pe('pointerup', { buttons: 0 }); me('mouseup', { buttons: 0 }); me('click', { buttons: 0 });
    if (dbl) { pe('pointerdown', { buttons: 1, detail: 2 }); me('mousedown', { buttons: 1, detail: 2 }); pe('pointerup', { buttons: 0, detail: 2 }); me('mouseup', { buttons: 0, detail: 2 }); me('click', { buttons: 0, detail: 2 }); me('dblclick', { buttons: 0, detail: 2 }); }
  }

  // React 속성/상태를 뒤져 RealGrid GridView(setValue·getColumns 있는 객체)를 찾는다
  function findGridApi(root) {
    const isView = (o) => !!o && typeof o === 'object' && typeof o.setValue === 'function' && typeof o.getColumns === 'function';
    const seen = new Set();
    const scan = (o, d) => {
      if (!o || typeof o !== 'object' || d > 4 || seen.has(o) || o instanceof Node || o instanceof Window) return null; seen.add(o);
      if (isView(o)) return o;
      let keys; try { keys = Object.keys(o); } catch (e) { return null; }
      if (keys.length > 60) return null;
      for (const k of keys) { if (/^(_reactInternals|__reactFiber|return|child|sibling|alternate|stateNode)$/.test(k)) continue; let v; try { v = o[k]; } catch (e) { continue; } const r = scan(v, d + 1); if (r) return r; }
      return null;
    };
    for (const k of Object.keys(root)) { if (!k.startsWith('__react')) { const r = scan(root[k], 0); if (r) return r; } }
    const fk = Object.keys(root).find((k) => k.startsWith('__reactFiber$'));
    for (let f = fk ? root[fk] : null, i = 0; f && i < 15; f = f.return, i++) {
      for (const src of [f.memoizedProps, f.memoizedState, f.stateNode]) { const r = scan(src, 0); if (r) return r; }
      // 훅 상태(useRef/useState)는 memoizedState가 이어진 목록이라 따로 따라간다
      for (let h = f.memoizedState, j = 0; h && typeof h === 'object' && j < 30; h = h.next, j++) { const r = scan(h.memoizedState, 0); if (r) return r; }
    }
    for (const k of Object.keys(window)) { try { if (isView(window[k])) return window[k]; } catch (e) {} }
    return null;
  }

        for (const c of cells) { if (!/rg-checked-row/.test(c.parentElement.className)) { pressOn(c); await sleep(80); } }
        if (checkedCount()) checkedBy = '줄별 체크';
      }
      if (!checkedBy) throw new Error('올린 줄을 선택(체크)하지 못했어요. 목록 왼쪽 체크칸을 모두 선택한 뒤 [쿠폰 상품 일괄등록]을 직접 눌러 주세요.');
      status = `줄 선택함(${checkedBy}). [쿠폰 상품 일괄등록]을 누르는 중…`; render();
      await sleep(300);
      const btn = textEl(/^쿠폰상품일괄등록$/, 'button, [role=button]')[0] || textEl(/일괄등록$/, 'button, [role=button]').filter((b) => b !== tab).pop();
      if (!btn) throw new Error('[쿠폰 상품 일괄등록] 버튼을 못 찾았어요. 줄은 선택해 두었으니 직접 눌러 주세요.');
      if (btn.disabled) throw new Error('[쿠폰 상품 일괄등록] 버튼이 꺼져 있어요. 줄은 선택해 두었으니 화면을 확인해 주세요.');
      dodge(btn);
      const r = await driveDialogs({ doneRe: /완료|등록되었|요청되었|성공/, failRe: /실패|오류|없습니다|잘못|형식/, trigger: () => press(btn), extra: handleContribDialog });
      const msg = r.done ? `상품-쿠폰 일괄 등록 완료 (${rows.length}줄). ${r.msgs.join(' / ')}` : r.fail ? `일괄 등록 실패: ${r.fail}` : `[쿠폰 상품 일괄등록]을 눌렀지만 완료 안내를 못 봤어요. 화면을 확인해 주세요. ${r.msgs.length ? '본 창: ' + r.msgs.join(' / ') : ''}`;
      store.set('bulk', { ok: r.done, msg, at: new Date().toISOString() }); status = r.done ? '일괄 등록 완료' : '일괄 등록 확인 필요';
    } catch (e) { store.set('bulk', { ok: false, msg: e.message }); status = '일괄 등록 중 멈췄어요.'; }
    finally { busy = false; render(); }
  }

  function confirmSaved() {
    const code = field('cpnNo')?.value?.trim();
    const v = prompt('세일즈원에 저장된 쿠폰코드를 확인해 주세요.', code || '');
    if (!v) return;
    if (!/^\d{6,}$/.test(v.trim())) { alert('쿠폰코드는 숫자로 입력해 주세요.'); return; }
    if (lastResult?.rows) { const fl = store.get('fills', {}); fl[current] = lastResult.rows.map(({ key, value, picked, ok, why }) => ({ key, value, picked, ok, why })); store.set('fills', fl); }
    setCode(current, v.trim()); current = null; lastResult = null; clearMarks(); render();
  }

  async function copy(t) {
    // 회사 PC는 클립보드 권한이 막혀 있을 수 있어 옛 방식(execCommand)부터 시도
    const ta = document.createElement('textarea'); ta.value = t; ta.style.cssText = 'position:fixed;left:-9999px'; document.body.appendChild(ta); ta.select();
    let ok = false; try { ok = document.execCommand('copy'); } catch {} ta.remove();
    if (ok) { alert(`${t.split('\n').length}줄 복사했어요.`); return; }
    try { await navigator.clipboard.writeText(t); alert(`${t.split('\n').length}줄 복사했어요.`); }
    catch { prompt('아래 내용을 복사해 주세요 (Ctrl+C).', t); }
  }
